# Right-Hand Racket Grip Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a standalone CPU MuJoCo right-hand forehand-grip pretraining pipeline for holding a badminton racket.

**Architecture:** Add a focused `src/grip/` package that can audit the hand/racket model, generate or validate a grip scene, solve a static reference grip, run a standalone grip-hold environment, and validate acceptance metrics. Keep the first version independent from full-body swing imitation and MJX/Warp training.

**Tech Stack:** Python 3.11, MuJoCo Python API, `mujoco.MjSpec`, NumPy, SciPy optimization, pytest, existing `musclemimic_models`, existing `environment/racket` assets.

Create `src/grip/target_config.py`:

```python
from __future__ import annotations

import json
import math
from dataclasses import dataclass
from pathlib import Path

import numpy as np

from src.grip.paths import target_config_path


@dataclass(frozen=True)
class GripTargetPoint:
    y: float
    theta_deg: float
    weight: float


@dataclass(frozen=True)
class GripTargetConfig:
    handle_radius_m: float
    contact_clearance_m: float
    target_points_racket_local: dict[str, GripTargetPoint]
    raw: dict

    def target_xyz(self, name: str) -> np.ndarray:
        point = self.target_points_racket_local[name]
        theta = math.radians(point.theta_deg)
        return np.array(
            [
                self.handle_radius_m * math.cos(theta),
                point.y,
                self.handle_radius_m * math.sin(theta),
            ],
            dtype=float,
        )

    def target_weight(self, name: str) -> float:
        return self.target_points_racket_local[name].weight


def load_grip_target_config(path: str | Path | None = None) -> GripTargetConfig:
    config_path = Path(path) if path is not None else target_config_path()
    with config_path.open("r", encoding="utf-8") as f:
        raw = json.load(f)
    points = {
        name: GripTargetPoint(
            y=float(value["y"]),
            theta_deg=float(value["theta_deg"]),
            weight=float(value["weight"]),
        )
        for name, value in raw["target_points_racket_local"].items()
    }
    required = {"palm", "thumb", "index", "middle", "ring", "pinky"}
    missing = required.difference(points)
    if missing:
        raise ValueError(f"missing grip target points: {sorted(missing)}")
    return GripTargetConfig(
        handle_radius_m=float(raw["handle_radius_m"]),
        contact_clearance_m=float(raw["contact_clearance_m"]),
        target_points_racket_local=points,
        raw=raw,
    )
```

@dataclass(frozen=True)
class HandRacketModelMap:
    hand_bodies: dict[str, str]
    hand_sites: dict[str, str]
    right_hand_joint_names: list[str]
    right_hand_actuator_names: list[str]
    racket_body: str | None
    racket_freejoint: str | None
    racket_sites: dict[str, str]
    handle_geoms: list[str]
    missing: list[str]

    @property
    def ok(self) -> bool:
        return not self.missing


def _name_exists(model: mujoco.MjModel, obj_type: mujoco.mjtObj, name: str) -> bool:
    return mujoco.mj_name2id(model, obj_type, name) >= 0


def _first_existing(model: mujoco.MjModel, obj_type: mujoco.mjtObj, candidates: list[str]) -> str | None:
    for name in candidates:
        if _name_exists(model, obj_type, name):
            return name
    return None


def _existing_names(model: mujoco.MjModel, obj_type: mujoco.mjtObj, count: int, names: list[str]) -> list[str]:
    return [name for name in names if _name_exists(model, obj_type, name)]


def load_model_map(
    model: mujoco.MjModel,
    *,
    require_racket: bool = True,
    require_grip_sites: bool = True,
) -> HandRacketModelMap:
    missing: list[str] = []
    hand_bodies = {}
    for group, candidates in RIGHT_HAND_BODY_CANDIDATES.items():
        body_name = _first_existing(model, mujoco.mjtObj.mjOBJ_BODY, candidates)
        if body_name is None:
            missing.append(f"body:{group}:{candidates}")
        else:
            hand_bodies[group] = body_name

    right_hand_joint_names = _existing_names(
        model,
        mujoco.mjtObj.mjOBJ_JOINT,
        model.njnt,
        RIGHT_HAND_JOINT_NAMES,
    )
    missing.extend(f"joint:{name}" for name in RIGHT_HAND_JOINT_NAMES if name not in right_hand_joint_names)

    right_hand_actuator_names = _existing_names(
        model,
        mujoco.mjtObj.mjOBJ_ACTUATOR,
        model.nu,
        RIGHT_HAND_ACTUATOR_CANDIDATES,
    )
    missing.extend(f"actuator:{name}" for name in RIGHT_HAND_ACTUATOR_CANDIDATES if name not in right_hand_actuator_names)

    hand_sites = {}
    if require_grip_sites:
        for group, site_name in HAND_SITE_NAMES.items():
            if _name_exists(model, mujoco.mjtObj.mjOBJ_SITE, site_name):
                hand_sites[group] = site_name
            else:
                missing.append(f"site:{site_name}")

    racket_body = _first_existing(model, mujoco.mjtObj.mjOBJ_BODY, ["racket", "racket_handle", "badminton_racket"])
    racket_freejoint = _first_existing(model, mujoco.mjtObj.mjOBJ_JOINT, ["racket_free"])
    racket_sites = {}
    handle_geoms = []
    if require_racket:
        if racket_body is None:
            missing.append("body:racket")
        if racket_freejoint is None:
            missing.append("joint:racket_free")
        for site_name in RACKET_SITE_NAMES:
            if _name_exists(model, mujoco.mjtObj.mjOBJ_SITE, site_name):
                racket_sites[site_name] = site_name
            else:
                missing.append(f"site:{site_name}")
        handle_geoms = _existing_names(model, mujoco.mjtObj.mjOBJ_GEOM, model.ngeom, ["handle_grip"])
        if not handle_geoms:
            missing.append("geom:handle_grip")

    return HandRacketModelMap(
        hand_bodies=hand_bodies,
        hand_sites=hand_sites,
        right_hand_joint_names=right_hand_joint_names,
        right_hand_actuator_names=right_hand_actuator_names,
        racket_body=racket_body,
        racket_freejoint=racket_freejoint,
        racket_sites=racket_sites,
        handle_geoms=handle_geoms,
        missing=missing,
    )


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--xml", required=True)
    parser.add_argument("--allow-missing-grip-sites", action="store_true")
    args = parser.parse_args()
    model = mujoco.MjModel.from_xml_path(str(Path(args.xml)))
    model_map = load_model_map(model, require_grip_sites=not args.allow_missing_grip_sites)
    if model_map.ok:
        print("PASS: all required hand/racket names resolved")
        return 0
    print("FAIL: unresolved hand/racket names")
    for item in model_map.missing:
        print(f"- {item}")
    return 1


if __name__ == "__main__":
    raise SystemExit(main())

def _add_site_if_missing(body: mujoco.MjsBody, name: str, pos: list[float], size: list[float]) -> None:
    for site in body.sites:
        if site.name == name:
            return
    body.add_site(
        name=name,
        type=mujoco.mjtGeom.mjGEOM_SPHERE,
        pos=pos,
        size=size,
        rgba=[0.1, 0.8, 1.0, 1.0],
        group=4,
    )

from __future__ import annotations

import math

import numpy as np


def normalized(vec: np.ndarray, eps: float = 1e-9) -> np.ndarray:
    norm = float(np.linalg.norm(vec))
    if norm < eps:
        return np.zeros_like(vec, dtype=float)
    return np.asarray(vec, dtype=float) / norm


def angle_between_vectors(a: np.ndarray, b: np.ndarray) -> float:
    ahat = normalized(a)
    bhat = normalized(b)
    dot = float(np.clip(np.dot(ahat, bhat), -1.0, 1.0))
    return math.degrees(math.acos(dot))

from __future__ import annotations

import numpy as np


def weighted_site_target_residuals(
    current_sites: dict[str, np.ndarray],
    target_sites: dict[str, np.ndarray],
    weights: dict[str, float],
) -> np.ndarray:
    residuals = []
    for name in sorted(target_sites):
        diff = np.asarray(current_sites[name], dtype=float) - np.asarray(target_sites[name], dtype=float)
        residuals.extend(float(weights.get(name, 1.0)) * diff)
    return np.asarray(residuals, dtype=float)


def mean_site_error(current_sites: dict[str, np.ndarray], target_sites: dict[str, np.ndarray]) -> float:
    distances = [
        float(np.linalg.norm(np.asarray(current_sites[name], dtype=float) - np.asarray(target_sites[name], dtype=float)))
        for name in sorted(target_sites)
    ]
    return float(np.mean(distances)) if distances else 0.0


def joint_limit_margin_cost(qpos_values: np.ndarray, ranges: np.ndarray, margin: float = 0.03) -> float:
    if ranges.size == 0:
        return 0.0
    lower = ranges[:, 0] + margin
    upper = ranges[:, 1] - margin
    below = np.maximum(lower - qpos_values, 0.0)
    above = np.maximum(qpos_values - upper, 0.0)
    return float(np.sum(below * below + above * above))

import argparse
import json
from pathlib import Path

import mujoco
import numpy as np
from scipy.optimize import least_squares

from src.grip.hand_racket_model_map import HAND_SITE_NAMES, load_model_map
from src.grip.paths import reference_json_path, scene_xml_path, target_config_path
from src.grip.target_config import load_grip_target_config


def _joint_qpos_indices(model: mujoco.MjModel, joint_names: list[str]) -> list[int]:
    indices = []
    for name in joint_names:
        jid = mujoco.mj_name2id(model, mujoco.mjtObj.mjOBJ_JOINT, name)
        if jid >= 0 and model.jnt_qposadr[jid] >= 0:
            indices.append(int(model.jnt_qposadr[jid]))
    return indices


def _site_positions(model: mujoco.MjModel, data: mujoco.MjData, names: dict[str, str]) -> dict[str, np.ndarray]:
    mujoco.mj_forward(model, data)
    out = {}
    for group, site_name in names.items():
        sid = mujoco.mj_name2id(model, mujoco.mjtObj.mjOBJ_SITE, site_name)
        out[group] = np.array(data.site_xpos[sid], dtype=float)
    return out


def _racket_target_positions(model: mujoco.MjModel, data: mujoco.MjData, config) -> dict[str, np.ndarray]:
    grip_id = mujoco.mj_name2id(model, mujoco.mjtObj.mjOBJ_SITE, "grip_pose_site")
    racket_body_id = model.site_bodyid[grip_id]
    xpos = np.array(data.xpos[racket_body_id], dtype=float)
    xmat = np.array(data.xmat[racket_body_id], dtype=float).reshape(3, 3)
    return {name: xpos + xmat @ config.target_xyz(name) for name in HAND_SITE_NAMES}


def solve_reference(
    xml: str | Path = scene_xml_path(),
    targets: str | Path = target_config_path(),
    out: str | Path = reference_json_path(),
    *,
    max_nfev: int = 200,
) -> dict:
    model = mujoco.MjModel.from_xml_path(str(xml))
    data = mujoco.MjData(model)
    model_map = load_model_map(model)
    if not model_map.ok:
        raise ValueError(f"unresolved model map names: {model_map.missing}")

    config = load_grip_target_config(targets)
    qpos_indices = _joint_qpos_indices(model, model_map.right_hand_joint_names)
    x0 = np.array(data.qpos[qpos_indices], dtype=float)

    def residual(x: np.ndarray) -> np.ndarray:
        data.qpos[qpos_indices] = x
        current = _site_positions(model, data, HAND_SITE_NAMES)
        target = _racket_target_positions(model, data, config)
        values = []
        for name in sorted(current):
            values.extend(config.target_weight(name) * (current[name] - target[name]))
        values.extend(0.05 * (x - x0))
        return np.asarray(values, dtype=float)

    if qpos_indices:
        result = least_squares(residual, x0, max_nfev=max_nfev, verbose=0)
        data.qpos[qpos_indices] = result.x
    mujoco.mj_forward(model, data)

    current = _site_positions(model, data, HAND_SITE_NAMES)
    target = _racket_target_positions(model, data, config)
    site_errors = {name: float(np.linalg.norm(current[name] - target[name])) for name in sorted(current)}
    mean_error = float(np.mean(list(site_errors.values())))
    payload = {
        "xml": str(xml),
        "keyframe_name": "right_hand_racket_grip_ref",
        "qpos": data.qpos.tolist(),
        "qvel": data.qvel.tolist(),
        "racket_freejoint_qpos": [],
        "right_hand_joint_names": model_map.right_hand_joint_names,
        "site_errors_m": site_errors,
        "objective_breakdown": {"mean_site_error_m": mean_error},
        "notes": "Static reference generated by right-hand site-target least squares.",
    }
    out_path = Path(out)
    out_path.parent.mkdir(parents=True, exist_ok=True)
    out_path.write_text(json.dumps(payload, indent=2), encoding="utf-8")
    return {"mean_site_error_m": mean_error, "out": str(out_path)}


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--xml", default=str(scene_xml_path()))
    parser.add_argument("--targets", default=str(target_config_path()))
    parser.add_argument("--out", default=str(reference_json_path()))
    args = parser.parse_args()
    result = solve_reference(args.xml, args.targets, args.out)
    print(json.dumps(result, indent=2))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

from __future__ import annotations

import argparse
import json
from pathlib import Path

import mujoco
import numpy as np

from src.grip.hand_racket_model_map import HAND_SITE_NAMES, load_model_map
from src.grip.paths import reference_json_path, scene_xml_path, target_config_path
from src.grip.solve_right_hand_racket_grip import _racket_target_positions, _site_positions
from src.grip.target_config import load_grip_target_config


class RightHandRacketGripEnv:
    def __init__(self, xml: str | Path, targets: str | Path, reference: str | Path):
        self.model = mujoco.MjModel.from_xml_path(str(xml))
        self.data = mujoco.MjData(self.model)
        self.model_map = load_model_map(self.model)
        if not self.model_map.ok:
            raise ValueError(f"unresolved model map names: {self.model_map.missing}")
        self.config = load_grip_target_config(targets)
        self.reference = json.loads(Path(reference).read_text(encoding="utf-8"))
        self.actuator_ids = [
            mujoco.mj_name2id(self.model, mujoco.mjtObj.mjOBJ_ACTUATOR, name)
            for name in self.model_map.right_hand_actuator_names
        ]
        self.actuator_ids = [idx for idx in self.actuator_ids if idx >= 0]
        self.action_size = len(self.actuator_ids)
        self.step_count = 0
        self.max_episode_steps = 500

    def reset(self) -> tuple[np.ndarray, dict]:
        mujoco.mj_resetData(self.model, self.data)
        qpos = np.asarray(self.reference["qpos"], dtype=float)
        qvel = np.asarray(self.reference["qvel"], dtype=float)
        self.data.qpos[: len(qpos)] = qpos
        self.data.qvel[: len(qvel)] = qvel
        self.step_count = 0
        mujoco.mj_forward(self.model, self.data)
        return self._observation(), self._info()

    def step(self, action: np.ndarray) -> tuple[np.ndarray, float, bool, bool, dict]:
        action = np.asarray(action, dtype=float)
        if action.shape != (self.action_size,):
            raise ValueError(f"action shape {action.shape} does not match ({self.action_size},)")
        for ctrl_idx, value in zip(self.actuator_ids, np.clip(action, -1.0, 1.0), strict=True):
            self.data.ctrl[ctrl_idx] = value
        for _ in range(10):
            mujoco.mj_step(self.model, self.data)
        self.step_count += 1
        info = self._info()
        reward_terms = self._reward_terms(info, action)
        info["reward_terms"] = reward_terms
        reward = float(sum(reward_terms.values()))
        terminated = bool(not np.isfinite(reward))
        truncated = self.step_count >= self.max_episode_steps
        return self._observation(), reward, terminated, truncated, info

    def _observation(self) -> np.ndarray:
        return np.concatenate(
            [
                np.asarray(self.data.qpos, dtype=float),
                np.asarray(self.data.qvel, dtype=float),
                np.asarray(self.data.ctrl[self.actuator_ids], dtype=float) if self.actuator_ids else np.zeros(0),
            ]
        )

    def _info(self) -> dict:
        current = _site_positions(self.model, self.data, HAND_SITE_NAMES)
        target = _racket_target_positions(self.model, self.data, self.config)
        site_errors = {name: float(np.linalg.norm(current[name] - target[name])) for name in sorted(current)}
        return {
            "site_errors_m": site_errors,
            "mean_site_error_m": float(np.mean(list(site_errors.values()))),
            "contact_count": int(self.data.ncon),
            "step_count": self.step_count,
        }

    def _reward_terms(self, info: dict, action: np.ndarray) -> dict[str, float]:
        return {
            "r_site_match": float(np.exp(-20.0 * info["mean_site_error_m"])),
            "r_contact": float(min(info["contact_count"], 4) / 4.0),
            "r_effort": float(-0.001 * np.sum(action * action)),
        }


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--xml", default=str(scene_xml_path()))
    parser.add_argument("--targets", default=str(target_config_path()))
    parser.add_argument("--reference", default=str(reference_json_path()))
    parser.add_argument("--smoke-test", action="store_true")
    args = parser.parse_args()
    env = RightHandRacketGripEnv(args.xml, args.targets, args.reference)
    obs, info = env.reset()
    print(f"reset obs={obs.shape} mean_site_error_m={info['mean_site_error_m']:.6f}")
    if args.smoke_test:
        obs, reward, terminated, truncated, info = env.step(np.zeros(env.action_size))
        print(f"step obs={obs.shape} reward={reward:.6f} terminated={terminated} truncated={truncated}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

from __future__ import annotations

import argparse

import numpy as np

from src.grip.paths import reference_json_path, scene_xml_path, target_config_path

def evaluate(xml, targets, reference, episodes: int = 3, steps: int = 1000) -> dict:
    env = RightHandRacketGripEnv(xml, targets, reference)
    returns = []
    final_errors = []
    for _ in range(episodes):
        _, _ = env.reset()
        total = 0.0
        info = {}
        for _ in range(steps):
            _, reward, terminated, truncated, info = env.step(np.zeros(env.action_size))
            total += reward
            if terminated or truncated:
                break
        returns.append(total)
        final_errors.append(info.get("mean_site_error_m", 0.0))
    return {
        "episodes": episodes,
        "steps": steps,
        "mean_return": float(np.mean(returns)),
        "mean_final_site_error_m": float(np.mean(final_errors)),
    }

    parser.add_argument("--targets", default=str(target_config_path()))
    parser.add_argument("--reference", default=str(reference_json_path()))
    parser.add_argument("--episodes", type=int, default=3)
    parser.add_argument("--steps", type=int, default=1000)
    args = parser.parse_args()
    print(evaluate(args.xml, args.targets, args.reference, args.episodes, args.steps))
    return 0

```python
from __future__ import annotations

import argparse
import json
import math
from pathlib import Path

import numpy as np

from src.grip.paths import reference_json_path, scene_xml_path, target_config_path
from src.grip.right_hand_racket_grip_env import RightHandRacketGripEnv


def validate_grip(xml, targets, reference, steps: int = 200) -> dict:
    env = RightHandRacketGripEnv(xml, targets, reference)
    _, reset_info = env.reset()
    initial_qpos = np.array(env.data.qpos, dtype=float)
    initial_pos = np.array(env.data.site_xpos[0], dtype=float)
    info = reset_info
    for _ in range(steps):
        _, _, terminated, truncated, info = env.step(np.zeros(env.action_size))
        if terminated or truncated:
            break
    final_pos = np.array(env.data.site_xpos[0], dtype=float)
    qpos_delta = float(np.linalg.norm(np.array(env.data.qpos, dtype=float) - initial_qpos))
    return {
        "mean_site_error_m": float(info["mean_site_error_m"]),
        "translation_drift_m": float(np.linalg.norm(final_pos - initial_pos)),
        "orientation_drift_deg": float(min(180.0, math.degrees(qpos_delta))),
        "contact_count": int(info["contact_count"]),
        "finite": bool(np.all(np.isfinite(env.data.qpos)) and np.all(np.isfinite(env.data.qvel))),
    }


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--xml", default=str(scene_xml_path()))
    parser.add_argument("--targets", default=str(target_config_path()))
    parser.add_argument("--reference", default=str(reference_json_path()))
    parser.add_argument("--steps", type=int, default=200)
    args = parser.parse_args()
    report = validate_grip(args.xml, args.targets, args.reference, args.steps)
    print(json.dumps(report, indent=2))
    return 0 if report["finite"] else 1


if __name__ == "__main__":
    raise SystemExit(main())
```
